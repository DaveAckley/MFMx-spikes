# --- Define PRIMARY_CMAKE_RUNNING cache variable ---
# It defaults to OFF if not passed via -DPRIMARY_CMAKE_RUNNING=1
if(NOT DEFINED PRIMARY_CMAKE_RUNNING)
  set(PRIMARY_CMAKE_RUNNING OFF CACHE BOOL "Is this the primary CMake run that generates FileIDs.h?")
  message(STATUS "AHAX: DEFAULTED PRIMARY_CMAKE_RUNNING TO OFF")
endif()

if(PRIMARY_CMAKE_RUNNING)
  message(STATUS "AHAX: PRIMARY CMAKE BUILD RUNNING")
else()
  message(STATUS "AHAX: NON-PRIMARY CMAKE BUILD RUNNING")
endif()

# --- Include shared/CMakeLists.txt to get SHARED_SOURCES and SHARED_INCLUDE_DIRS ---
message(STATUS "AHAX: INCLUDING ${CMAKE_SOURCE_DIR}/srcs/shared/SharedDefinitions.cmake")  
include(${CMAKE_SOURCE_DIR}/srcs/shared/SharedDefinitions.cmake)
message(STATUS "AHAX: SHARED_INCLUDE_DIRS=${SHARED_INCLUDE_DIRS}")
#message(STATUS "AHAX: SHARED_SOURCES=${SHARED_SOURCES}")

# --- Handle EXTERNAL_FILE_IDS_HEADER_DIR for non-primary builds ---
# This variable will be passed by the pip build to point to the primary build's output.
if(NOT DEFINED EXTERNAL_FILE_IDS_HEADER_DIR)
  set(EXTERNAL_FILE_IDS_HEADER_DIR "" CACHE PATH "Path to directory containing externally generated FileIDs.h")
endif()

# Include the function definition
include(${CMAKE_SOURCE_DIR}/cmake/GenerateFileIDs.cmake)

# Call the function to generate/locate FileIDs.h
generate_file_ids_header(FileIDs.h)

# At this point, FILE_IDS_HEADER_DIR_FOR_THIS_BUILD should have been set by the function via PARENT_SCOPE.
if(NOT "${FILE_IDS_HEADER_DIR_FOR_THIS_BUILD}" STREQUAL "")
  message(STATUS "AHAX: FILE_IDS_HEADER_DIR_FOR_THIS_BUILD = ${FILE_IDS_HEADER_DIR_FOR_THIS_BUILD}")
  # Add this directory to the include paths for ALL targets in this build
  include_directories("${FILE_IDS_HEADER_DIR_FOR_THIS_BUILD}")
else()
  message(FATAL_ERROR "FILE_IDS_HEADER_DIR_FOR_THIS_BUILD is empty after calling generate_file_ids_header. This indicates an issue with the function or its invocation.")
endif()

# Store the path to the generated header's directory in a global property
# This is how the primary build makes its generated header location known to external builds.
if(PRIMARY_CMAKE_RUNNING)
  #set_property(GLOBAL PROPERTY CANONICAL_FILE_IDS_HEADER_DIR "${FILE_IDS_HEADER_DIR_FOR_THIS_BUILD}")
  set(CANONICAL_FILE_IDS_HEADER_DIR_CACHE "${FILE_IDS_HEADER_DIR_FOR_THIS_BUILD}"
    CACHE PATH "Canonical directory where FileIDs.h is generated by the primary build." FORCE)
  message(STATUS "AHAX: GLOBAL PROPERTY CANONICAL_FILE_IDS_HEADER_DIR_CACHE ${CANONICAL_FILE_IDS_HEADER_DIR_CACHE}")
  add_subdirectory(srcs/cross)
else()
  message(STATUS "AHAX: HOST BUILD")  
  set(PYBIND11_FINDPYTHON ON)
  set(pybind11_DIR ../../venv/lib/python3.10/site-packages/pybind11/share/cmake/pybind11)
  find_package(pybind11 CONFIG REQUIRED)

  # Add the source subdirectory
  add_subdirectory(srcs/host)

endif()
